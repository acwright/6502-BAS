REM ============================================
REM  PROG1.BAS: PRINT, REM, END
REM  Tests string output, expression printing,
REM  semicolons, commas, and END statement.
REM ============================================
10 REM THIS IS A COMMENT
20 PRINT "HELLO WORLD"
30 PRINT 2+3
40 PRINT "A";
50 PRINT "B"
60 PRINT "X", "Y"
70 END

REM ============================================
REM  PROG2.BAS: LET, VARIABLES, ARITHMETIC
REM  Tests variable assignment, +, -, *, /
REM ============================================
10 LET A=10
20 B=25
30 C=A+B
40 D=B-A
50 E=A*3
60 F=B/5
70 PRINT "A+B=";C
80 PRINT "B-A=";D
90 PRINT "A*3=";E
100 PRINT "B/5=";F

REM ============================================
REM  PROG3.BAS: IF, THEN, GOTO, COMPARISONS
REM  Tests IF...THEN, =, <>, <, >, <=, >=
REM  and GOTO with implicit line number.
REM ============================================
10 A=5
20 IF A=5 THEN PRINT "A=5 OK"
30 IF A<>3 THEN PRINT "A<>3 OK"
40 IF A>2 THEN PRINT "A>2 OK"
50 IF A<10 THEN PRINT "A<10 OK"
60 IF A>=5 THEN PRINT "A>=5 OK"
70 IF A<=5 THEN PRINT "A<=5 OK"
80 IF A=5 THEN 100
90 PRINT "SHOULD NOT PRINT"
100 PRINT "GOTO VIA THEN OK"

REM ============================================
REM  PROG4.BAS: FOR, TO, STEP, NEXT
REM  Tests FOR/NEXT loops with default step
REM  and explicit STEP values.
REM ============================================
10 PRINT "COUNT 1-5:"
20 FOR I=1 TO 5
30 PRINT I;
40 NEXT I
50 PRINT
60 PRINT "EVEN 2-10:"
70 FOR J=2 TO 10 STEP 2
80 PRINT J;
90 NEXT J
100 PRINT
110 PRINT "DOWN 5-1:"
120 FOR K=5 TO 1 STEP -1
130 PRINT K;
140 NEXT K
150 PRINT

REM ============================================
REM  PROG5.BAS: GOSUB, RETURN
REM  Tests subroutine calling and return.
REM ============================================
10 PRINT "BEFORE GOSUB"
20 GOSUB 50
30 PRINT "AFTER GOSUB"
40 END
50 PRINT "IN SUBROUTINE"
60 RETURN

REM ============================================
REM  PROG8.BAS: LOGIC OPERATORS
REM  Tests AND, OR, NOT in expressions
REM  and conditions.
REM ============================================
10 A=1:B=0
20 IF A AND NOT B THEN PRINT "AND/NOT OK"
30 IF A OR B THEN PRINT "OR OK"
40 IF NOT B THEN PRINT "NOT OK"
50 PRINT "5 AND 3=";5 AND 3
60 PRINT "5 OR 2=";5 OR 2
70 PRINT "NOT 0=";NOT 0

REM ============================================
REM  PROG9.BAS: ON...GOTO, ON...GOSUB
REM  Tests computed branching.
REM ============================================
10 N=2
20 ON N GOTO 40,50,60
30 PRINT "FALLTHROUGH":GOTO 70
40 PRINT "FIRST":GOTO 70
50 PRINT "SECOND":GOTO 70
60 PRINT "THIRD":GOTO 70
70 ON 1 GOSUB 90
80 END
90 PRINT "ON GOSUB OK"
100 RETURN

REM ============================================
REM  PROG10.BAS: DATA, READ, RESTORE
REM  Tests reading inline data values and
REM  restarting the data pointer.
REM ============================================
10 DATA 10,20,30
20 READ A:READ B:READ C
30 PRINT A;B;C
40 RESTORE
50 READ D
60 PRINT "RESTORED: ";D

REM ============================================
REM  PROG11.BAS: DIM (ARRAYS)
REM  Tests 1D array allocation, writing,
REM  and reading.
REM ============================================
10 DIM A(5)
20 FOR I=0 TO 5
30 A(I)=I*I
40 NEXT I
50 PRINT "SQUARES 0-5:"
60 FOR I=0 TO 5
70 PRINT A(I);
80 NEXT I
90 PRINT

REM ============================================
REM  PROG13.BAS: PEEK, POKE, CLR
REM  Tests memory read/write and CLR to
REM  reset variables.
REM ============================================

20 PRINT "A=";A
30 POKE 1024,170
40 PRINT "PEEK=";PEEK(1024)
50 CLR
60 PRINT "AFTER CLR A=";A

REM ============================================
REM  PROG15.BAS: STOP, MEM, INKEY, PAUSE
REM  Tests STOP/CONT flow, memory info,
REM  key polling, and timed pause.
REM ============================================
10 MEM
20 PRINT "PAUSING 1 SEC..."
30 PAUSE 100
40 PRINT "PRESS A KEY..."
50 K=INKEY
60 IF K=0 THEN 50
70 PRINT "KEY=";K
80 STOP
90 PRINT "CONT REACHED HERE"
